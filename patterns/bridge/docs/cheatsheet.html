Проблема: две независимые оси расширения

В умном доме есть разные виды управления: обычный пульт, расширенный пульт, пульт для презентаций. Есть разные виды устройств: телевизор, проектор, радио.

Если делать отдельный класс для каждой комбинации, появятся BasicTVRemote, AdvancedTVRemote, BasicProjectorRemote, AdvancedProjectorRemote и так далее. Добавление одного устройства потребует новых классов для всех видов пультов.

Решение и роли

Пульт хранит ссылку на MediaDevice и делегирует ему низкоуровневые операции. Иерархия пультов и иерархия устройств развиваются независимо. Здесь «реализация» означает сторону устройств, а не конкретный подкласс пульта.

РольУчастник
AbstractionRemote — включение и установка громкости
RefinedAbstractionAdvancedRemote — дополнительная операция mute()
ImplementorMediaDevice — контракт устройств
ConcreteImplementorTV, Projector
Remote ───── хранит ссылку ─────→ MediaDevice
  ↑                                ↑
AdvancedRemote                 TV, Projector

Оба пульта работают с обоими устройствами. Телевизор хранит громкость точно в процентах; проектор имеет 10 ступеней и округляет значение вниз до десятков. Например, команда 37% даёт 37% на телевизоре и 30% на проекторе. Контракт MediaDevice явно допускает это округление.

tv_remote = AdvancedRemote(TV())
projector_remote = AdvancedRemote(Projector())
tv_remote.mute()
projector_remote.mute()

Преимущество: новый пульт не требует новых подклассов для каждого устройства. Недостаток: появляются дополнительный интерфейс и делегирование; при одной неизменяемой комбинации это может быть лишней сложностью.

Важно объяснять назначение паттерна: сама по себе ссылка одного объекта на другой ещё не доказывает, что в программе применён Bridge.

Самостоятельное задание

В remotes.py добавьте два класса, не меняя существующие пульты и устройства:

  1. Radio(MediaDevice): изначально выключено, громкость 30. Реализуйте set_power(enabled), свойство volume и set_volume(percent). Радио хранит громкость точно в процентах; значения вне 0–100 вызывают ValueError и не изменяют предыдущее состояние. Обеспечьте поле is_on.
  2. PresentationRemote(Remote): метод start_presentation() включает переданное устройство и устанавливает громкость 0 через интерфейс MediaDevice. Повторный вызов должен оставлять устройство включённым с нулевой громкостью.

Новый вид устройства должен работать с существующими пультами, а новый пульт — с существующими устройствами. Проверьте и сочетание двух новых классов.

Реализация: remotes.py
"""Bridge: виды пультов и виды устройств развиваются независимо.

Remote — Abstraction; AdvancedRemote — RefinedAbstraction;
MediaDevice — Implementor; TV, Projector — ConcreteImplementor.
"""

from abc import ABC, abstractmethod


class MediaDevice(ABC):
    def __init__(self) -> None:
        self.is_on = False

    @abstractmethod
    def set_power(self, enabled: bool) -> None:
        """Установить питание."""

    @property
    @abstractmethod
    def volume(self) -> int:
        """Текущая громкость в процентах."""

    @abstractmethod
    def set_volume(self, percent: int) -> None:
        """Установить громкость 0–100; устройство может округлять её вниз."""


class TV(MediaDevice):
    def __init__(self) -> None:
        super().__init__()
        self._volume = 30

    def set_power(self, enabled: bool) -> None:
        self.is_on = enabled

    @property
    def volume(self) -> int:
        return self._volume

    def set_volume(self, percent: int) -> None:
        if not 0 <= percent <= 100:
            raise ValueError("Громкость должна быть от 0 до 100")
        self._volume = percent


class Projector(MediaDevice):
    def __init__(self) -> None:
        super().__init__()
        self._volume_steps = 3

    def set_power(self, enabled: bool) -> None:
        self.is_on = enabled

    @property
    def volume(self) -> int:
        return self._volume_steps * 10

    def set_volume(self, percent: int) -> None:
        if not 0 <= percent <= 100:
            raise ValueError("Громкость должна быть от 0 до 100")
        self._volume_steps = percent // 10


class Remote:
    def __init__(self, device: MediaDevice) -> None:
        self.device = device

    def toggle_power(self) -> None:
        self.device.set_power(not self.device.is_on)

    def set_volume(self, percent: int) -> None:
        self.device.set_volume(percent)


class AdvancedRemote(Remote):
    def mute(self) -> None:
        self.device.set_volume(0)

    remote_types = [remotes.Remote, remotes.AdvancedRemote]

    if radio is not None:
        device_types.append(radio)
    if presentation_remote is not None:
        remote_types.append(presentation_remote)
    for device_type in device_types:
        for remote_type in remote_types:
            device = device_type()
            remote = remote_type(device)
            remote.toggle_power()
            remote.set_volume(37)
            print(f"{remote_type.__name__} + {device_type.__name__}: питание={device.is_on}, звук={device.volume}%")

import unittest

from .. import remotes


class BridgeTests(unittest.TestCase):
    def test_all_remote_and_device_combinations(self):
        for remote_type in (remotes.Remote, remotes.AdvancedRemote):
            for device_type in (remotes.TV, remotes.Projector):
                with self.subTest(remote=remote_type.__name__, device=device_type.__name__):
                    device = device_type()
                    remote = remote_type(device)
                    self.assertIs(remote.device, device)
                    self.assertFalse(device.is_on)
                    remote.toggle_power()
                    self.assertTrue(device.is_on)
                    remote.set_volume(37)
                    self.assertEqual(device.volume, 37 if device_type is remotes.TV else 30)
                    remote.toggle_power()
                    self.assertFalse(device.is_on)

    def test_mute_works_with_every_implementation(self):
        for device_type in (remotes.TV, remotes.Projector):
            device = device_type()
            remote = remotes.AdvancedRemote(device)
            remote.toggle_power()
            remote.set_volume(80)
            remote.mute()
            self.assertEqual(device.volume, 0)
            self.assertTrue(device.is_on)

    def test_devices_check_volume_bounds(self):
        for device_type in (remotes.TV, remotes.Projector):
            device = device_type()
            for volume in (0, 100):
                device.set_volume(volume)
                self.assertEqual(device.volume, volume)
            for volume in (-1, 101):
                with self.assertRaises(ValueError):
                    device.set_volume(volume)
                self.assertEqual(device.volume, 100)

import unittest

from .. import remotes


class BridgeTaskTests(unittest.TestCase):
    def require(self, name):
        cls = getattr(remotes, name, None)
        self.assertIsNotNone(
            cls, f"САМОСТОЯТЕЛЬНО: добавьте {name} в remotes.py (см. TASK.md)"
        )
        return cls

    def test_radio_works_with_existing_remotes(self):
        radio_type = self.require("Radio")
        self.assertTrue(issubclass(radio_type, remotes.MediaDevice))
        for remote_type in (remotes.Remote, remotes.AdvancedRemote):
            radio = radio_type()
            self.assertFalse(radio.is_on)
            self.assertEqual(radio.volume, 30)
            remote = remote_type(radio)
            remote.toggle_power()
            self.assertTrue(radio.is_on)
            remote.set_volume(37)
            self.assertEqual(radio.volume, 37)
            remote.toggle_power()
            self.assertFalse(radio.is_on)
            if isinstance(remote, remotes.AdvancedRemote):
                remote.mute()
                self.assertEqual(radio.volume, 0)

    def test_radio_validates_volume(self):
        radio = self.require("Radio")()
        for volume in (0, 100):
            radio.set_volume(volume)
            self.assertEqual(radio.volume, volume)
        for volume in (-1, 101):
            with self.assertRaises(ValueError):
                radio.set_volume(volume)
            self.assertEqual(radio.volume, 100)

    def test_presentation_remote_works_with_existing_devices(self):
        remote_type = self.require("PresentationRemote")
        self.assertTrue(issubclass(remote_type, remotes.Remote))
        for device_type in (remotes.TV, remotes.Projector):
            device = device_type()
            remote = remote_type(device)
            remote.set_volume(70)
            remote.start_presentation()
            remote.start_presentation()
            self.assertTrue(device.is_on)
            self.assertEqual(device.volume, 0)
            remote.toggle_power()
            self.assertFalse(device.is_on)

    def test_new_remote_and_new_device_work_together(self):
        radio = self.require("Radio")()
        remote = self.require("PresentationRemote")(radio)
        remote.start_presentation()
        self.assertTrue(radio.is_on)
        self.assertEqual(radio.volume, 0)
