Facade — Фасад

Проблема

Чтобы посмотреть фильм, клиент должен закрыть шторы, приглушить свет, включить проектор с правильным источником и настроить звук. Без фасада каждый клиент повторяет последовательность и знает интерфейсы четырёх подсистем:

curtains.close()
lighting.set_brightness(20)
projector.turn_on("HDMI 1")
sound.turn_on(35)

Решение и роли

HomeFacade.start_movie() выражает намерение пользователя одним вызовом. Метод end_movie() выключает проектор и звук, возвращает яркость 100% и открывает шторы. Он устанавливает заранее определённое состояние, а не восстанавливает произвольные прежние настройки.

РольУчастник
Facade — высокоуровневый интерфейсHomeFacade
ПодсистемыLighting, Curtains, Projector, SoundSystem
Clientкод демонстрации, вызывающий home.start_movie()
Клиент → HomeFacade.start_movie()
           ├─ Curtains.close()
           ├─ Lighting.set_brightness(20)
           ├─ Projector.turn_on("HDMI 1")
           └─ SoundSystem.turn_on(35)

Фасад получает экземпляры подсистем через конструктор. Подсистемы не знают о фасаде; их по-прежнему можно использовать напрямую для нестандартных сценариев.

Код и запуск

  • home.py — четыре подсистемы и фасад.
  • __main__.py — начало и завершение просмотра фильма.

Нужен Python 3.10+ и pytest 9+. В Codespaces pytest устанавливается автоматически. Из этой папки паттерна (её можно передать студенту отдельно):

Применимость и ограничения

Фасад полезен, когда клиентам нужен небольшой набор типовых операций сложной подсистемы. Он уменьшает число зависимостей клиента и собирает порядок действий в одном месте. Если включить в него все возможные сценарии, класс может стать слишком большим. Фасад сам по себе не обеспечивает откат при сбое одной из подсистем.

Вопрос: чем start_movie() отличается по назначению от LampObjectAdapter.set_power()?

  • яркость освещения равна 0;

Реализация: home.py
"""Facade: один вызов настраивает подсистемы для просмотра фильма.

HomeFacade — Facade; Lighting, Curtains, Projector, SoundSystem — подсистемы.
Подсистемы не знают о фасаде и доступны клиенту напрямую.
"""


class Lighting:
    def __init__(self) -> None:
        self.brightness = 100

    def set_brightness(self, percent: int) -> None:
        if not 0 <= percent <= 100:
            raise ValueError("Яркость должна быть от 0 до 100")
        self.brightness = percent


class Curtains:
    def __init__(self) -> None:
        self.is_closed = False

    def close(self) -> None:
        self.is_closed = True

    def open(self) -> None:
        self.is_closed = False


class Projector:
    def __init__(self) -> None:
        self.is_on = False
        self.source: str | None = None

    def turn_on(self, source: str) -> None:
        self.source = source
        self.is_on = True

    def turn_off(self) -> None:
        self.is_on = False
        self.source = None


class SoundSystem:
    def __init__(self) -> None:
        self.is_on = False
        self.volume = 0

    def turn_on(self, volume: int) -> None:
        if not 0 <= volume <= 100:
            raise ValueError("Громкость должна быть от 0 до 100")
        self.volume = volume
        self.is_on = True

    def turn_off(self) -> None:
        self.is_on = False
        self.volume = 0


class HomeFacade:
    def __init__(
        self,
        lighting: Lighting,
        curtains: Curtains,
        projector: Projector,
        sound: SoundSystem,
    ) -> None:
        self.lighting = lighting
        self.curtains = curtains
        self.projector = projector
        self.sound = sound

    def start_movie(self) -> None:
        self.curtains.close()
        self.lighting.set_brightness(20)
        self.projector.turn_on("HDMI 1")
        self.sound.turn_on(35)

    def end_movie(self) -> None:
        self.projector.turn_off()
        self.sound.turn_off()
        self.lighting.set_brightness(100)
        self.curtains.open()

    # ЗАДАНИЕ: реализуйте HomeFacade.leave_home() — сценарий «Ушёл из дома».
    # Используйте публичные методы подсистем.
    # Условия и команды проверки: TASK.md; тесты: tests/test_task.py.
Демонстрация: __main__.py
if __package__:
    from .home import Curtains, HomeFacade, Lighting, Projector, SoundSystem
else:
    from home import Curtains, HomeFacade, Lighting, Projector, SoundSystem


def demo() -> None:
    print("=== Facade: один вызов вместо настройки четырёх подсистем ===")
    lighting, curtains = Lighting(), Curtains()
    home = HomeFacade(lighting, curtains, Projector(), SoundSystem())
    home.start_movie()
    print(
        f"Кино: свет={home.lighting.brightness}%, шторы закрыты={home.curtains.is_closed}, "
        f"проектор={home.projector.source}, звук={home.sound.volume}%"
    )
    lighting.set_brightness(70)
    print(f"Прямой вызов подсистемы: свет={lighting.brightness}%")
    home.end_movie()
    print(f"Фильм окончен: свет={home.lighting.brightness}%, проектор={home.projector.is_on}")
    leave_home = getattr(home, "leave_home", None)
    if leave_home is not None:
        leave_home()
        print(f"Ушли из дома: свет={home.lighting.brightness}%, шторы закрыты={home.curtains.is_closed}")


if __name__ == "__main__":
    demo()

Тесты: tests/test_example.py
import unittest
from unittest.mock import Mock

from ..home import Curtains, HomeFacade, Lighting, Projector, SoundSystem


def make_home():
    return HomeFacade(Lighting(), Curtains(), Projector(), SoundSystem())


class FacadeTests(unittest.TestCase):
    def test_start_movie_configures_all_subsystems(self):
        home = make_home()
        home.start_movie()
        self.assertEqual(home.lighting.brightness, 20)
        self.assertTrue(home.curtains.is_closed)
        self.assertTrue(home.projector.is_on)
        self.assertEqual(home.projector.source, "HDMI 1")
        self.assertTrue(home.sound.is_on)
        self.assertEqual(home.sound.volume, 35)

    def test_end_movie_resets_all_subsystems(self):
        home = make_home()
        home.start_movie()
        home.end_movie()
        self.assertEqual(home.lighting.brightness, 100)
        self.assertFalse(home.curtains.is_closed)
        self.assertFalse(home.projector.is_on)
        self.assertIsNone(home.projector.source)
        self.assertFalse(home.sound.is_on)
        self.assertEqual(home.sound.volume, 0)

    def test_subsystems_remain_available_without_facade(self):
        home = make_home()
        home.lighting.set_brightness(70)
        self.assertEqual(home.lighting.brightness, 70)
        self.assertFalse(home.projector.is_on)

    def test_invalid_levels_do_not_change_subsystems(self):
        light = Lighting()
        sound = SoundSystem()
        for value in (-1, 101):
            with self.subTest(value=value):
                with self.assertRaises(ValueError):
                    light.set_brightness(value)
                with self.assertRaises(ValueError):
                    sound.turn_on(value)
        self.assertEqual(light.brightness, 100)
        self.assertEqual(sound.volume, 0)
        self.assertFalse(sound.is_on)
