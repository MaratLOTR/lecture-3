Пример из презентации «Лекция 3»: слайды 7–16; практика: слайд 19. Номера соответствуют фактическому порядку слайдов в файле.

Проблема

Приложение управляет приборами через set_power(enabled). ModernLight уже поддерживает этот интерфейс. Готовая LegacyLamp предоставляет только switch_on() и switch_off(). Если передать её клиенту напрямую, получим AttributeError: у старой лампы нет set_power().

Adapter преобразует готовый интерфейс в тот, которого ожидает клиент. Старую библиотеку и общий алгоритм управления менять не требуется.

Адаптер класса явно вызывает LegacyLamp.__init__(self). Он предоставляет и унаследованные switch_on() / switch_off(), и новый set_power(). Адаптер объекта делегирует команды одному и тому же переданному объекту.

Применимость и ограничения

Adapter нужен при подключении стороннего или устаревшего API. Он позволяет повторно использовать код и изолировать особенности этого API от клиента. Цена — дополнительные классы и делегирование; изменение API потребует проверки адаптера. Для простого случая отдельный класс может оказаться избыточным. Адаптация должна сохранять смысл операций, даже если различаются единицы измерения.

Вопрос: какой вариант позволяет подключить уже созданный экземпляр лампы и почему?

Задание: два адаптера вентилятора

Практика со слайда 19 презентации «Лекция 3». Номер — фактический порядок слайда в файле. Работайте в devices.py. Примеры адаптеров лампы уже реализованы.

Готовый ForeignFan имеет метод set_mode("RUN") / set_mode("STOP"). Клиент ожидает Switchable.set_power(enabled).

  1. Добавьте FanAdapter(Switchable) — адаптер объекта. Конструктор принимает готовый ForeignFan и сохраняет именно этот экземпляр в self.fan. set_power(True) вызывает self.fan.set_mode("RUN"), set_power(False) — self.fan.set_mode("STOP").
  2. Добавьте FanClassAdapter(Switchable, ForeignFan) — адаптер класса. Конструктор без аргументов инициализирует унаследованный ForeignFan; начальный mode равен "STOP". set_power() вызывает унаследованный self.set_mode() с теми же значениями. Старый метод остаётся доступен.
  3. Проверьте оба адаптера в одном списке с ModernLight, LampObjectAdapter и LampClassAdapter. Весь список управляется неизменённым set_all_power(). Повторные команды включения и выключения также должны работать.

Не меняйте ForeignFan, Switchable, set_all_power() и тесты. Не присваивайте mode напрямую: используйте готовый API вентилятора. После добавления классов демонстрация автоматически покажет оба вентилятора.

Реализация: devices.py
"""Adapter: клиент управляет новой и старой лампами через один интерфейс.

Switchable — Target; LegacyLamp — Adaptee;
LampObjectAdapter, LampClassAdapter — Adapter; set_all_power — Client.
"""

from abc import ABC, abstractmethod
from collections.abc import Iterable


class Switchable(ABC):
    @abstractmethod
    def set_power(self, enabled: bool) -> None:
        """Установить питание устройства."""


class ModernLight(Switchable):
    def __init__(self) -> None:
        self.is_on = False

    def set_power(self, enabled: bool) -> None:
        self.is_on = enabled


class LegacyLamp:
    """Имитирует готовую стороннюю библиотеку; её интерфейс не меняем."""

    def __init__(self) -> None:
        self.powered = False

    def switch_on(self) -> None:
        self.powered = True

    def switch_off(self) -> None:
        self.powered = False


class LampObjectAdapter(Switchable):
    def __init__(self, lamp: LegacyLamp) -> None:
        self.lamp = lamp

    def set_power(self, enabled: bool) -> None:
        if enabled:
            self.lamp.switch_on()
        else:
            self.lamp.switch_off()


class LampClassAdapter(Switchable, LegacyLamp):
    def __init__(self) -> None:
        LegacyLamp.__init__(self)

    def set_power(self, enabled: bool) -> None:
        if enabled:
            self.switch_on()
        else:
            self.switch_off()


# Сохраняем прежнее имя для уже написанных примеров.
LegacyLampAdapter = LampObjectAdapter


def set_all_power(devices: Iterable[Switchable], enabled: bool) -> None:
    """Клиенту не нужно знать конкретные классы и старые методы."""
    for device in devices:
        device.set_power(enabled)


class ForeignFan:
    """Ещё один сторонний API, который предстоит адаптировать."""

    def __init__(self) -> None:
        self.mode = "STOP"

    def set_mode(self, mode: str) -> None:
        if mode not in ("RUN", "STOP"):
            raise ValueError("Допустимые режимы: RUN и STOP")
        self.mode = mode

    light = devices.ModernLight()
    lamp = devices.LegacyLamp()
    object_adapter = devices.LampObjectAdapter(lamp)
    class_adapter = devices.LampClassAdapter()

from abc import ABC, abstractmethod


class TemperatureControl(ABC):
    @abstractmethod
    def get_celsius(self) -> float:
        """Прочитать температуру в °C."""

    @abstractmethod
    def set_celsius(self, temperature: float) -> None:
        """Установить температуру в °C."""


class ForeignThermostat:
    """Готовый сторонний API; менять не нужно."""

    def __init__(self) -> None:
        self._fahrenheit = 68.0

    def read_fahrenheit(self) -> float:
        return self._fahrenheit

    def write_fahrenheit(self, temperature: float) -> None:
        if not 50 <= temperature <= 86:
            raise ValueError("Температура должна быть от 50 до 86 °F")
        self._fahrenheit = temperature

import unittest
from unittest.mock import patch

from .. import devices


class AdapterTests(unittest.TestCase):
    def test_same_client_controls_modern_and_legacy_lights(self):
        light = devices.ModernLight()
        lamp = devices.LegacyLamp()
        adapter = devices.LampObjectAdapter(lamp)
        self.assertIs(adapter.lamp, lamp)
        for enabled in (True, False, False, True):
            devices.set_all_power([light, adapter], enabled)
            self.assertEqual(light.is_on, enabled)
            self.assertEqual(lamp.powered, enabled)

    def test_adapter_delegates_to_legacy_api(self):
        lamp = devices.LegacyLamp()
        adapter = devices.LampObjectAdapter(lamp)
        with patch.object(lamp, "switch_on", wraps=lamp.switch_on) as on:
            adapter.set_power(True)
            on.assert_called_once_with()
        with patch.object(lamp, "switch_off", wraps=lamp.switch_off) as off:
            adapter.set_power(False)
            off.assert_called_once_with()

    def test_foreign_fan_rejects_unknown_mode(self):
        fan = devices.ForeignFan()
        with self.assertRaises(ValueError):
            fan.set_mode("FAST")
        self.assertEqual(fan.mode, "STOP")

    def test_class_adapter_implements_both_interfaces(self):
        adapter = devices.LampClassAdapter()
        self.assertIsInstance(adapter, devices.Switchable)
        self.assertIsInstance(adapter, devices.LegacyLamp)
        self.assertFalse(adapter.powered)
        adapter.switch_on()
        self.assertTrue(adapter.powered)
        adapter.switch_off()
        self.assertFalse(adapter.powered)
        light = devices.ModernLight()
        lamp = devices.LegacyLamp()
        for enabled in (True, True, False, False, True):
            devices.set_all_power([light, devices.LampObjectAdapter(lamp), adapter], enabled)
            self.assertEqual((light.is_on, lamp.powered, adapter.powered), (enabled,) * 3)

    def test_class_adapter_uses_inherited_api(self):
        adapter = devices.LampClassAdapter()
        for enabled, method in ((True, "switch_on"), (False, "switch_off")):
            with self.subTest(enabled=enabled):
                with patch.object(adapter, method, wraps=getattr(adapter, method)) as call:
                    adapter.set_power(enabled)
                    call.assert_called_once_with()

    def test_old_adapter_name_remains_available(self):
        self.assertIs(devices.LegacyLampAdapter, devices.LampObjectAdapter)

class FanAdapterTaskTests(unittest.TestCase):

    def test_object_adapter_wraps_original_fan(self):

        self.assertIs(adapter.fan, fan)
        for enabled, mode in ((True, "RUN"), (True, "RUN"), (False, "STOP"), (False, "STOP")):

    def test_object_adapter_calls_foreign_api(self):

import unittest
from unittest.mock import patch

from ..adapter_thermostat import ForeignThermostat, ThermostatAdapter


class ThermostatTests(unittest.TestCase):
    def test_reads_the_current_temperature_of_original_device(self):
        thermostat = ForeignThermostat()
        adapter = ThermostatAdapter(thermostat)
        self.assertAlmostEqual(adapter.get_celsius(), 20)
        thermostat.write_fahrenheit(86)
        self.assertAlmostEqual(adapter.get_celsius(), 30)

    def test_writes_fractional_temperature_through_foreign_api(self):
        thermostat = ForeignThermostat()
        adapter = ThermostatAdapter(thermostat)
        with patch.object(thermostat, "write_fahrenheit", wraps=thermostat.write_fahrenheit) as write:
            adapter.set_celsius(22.5)
            write.assert_called_once()
            self.assertAlmostEqual(write.call_args.args[0], 72.5)
        self.assertAlmostEqual(thermostat.read_fahrenheit(), 72.5)
        self.assertAlmostEqual(adapter.get_celsius(), 22.5)

    def test_boundaries_and_rejected_commands(self):
        thermostat = ForeignThermostat()
        adapter = ThermostatAdapter(thermostat)
        for celsius, fahrenheit in ((10, 50), (30, 86)):
            adapter.set_celsius(celsius)
            self.assertAlmostEqual(thermostat.read_fahrenheit(), fahrenheit)
        for invalid in (9, 31, float("nan")):
            with self.assertRaises(ValueError):
                adapter.set_celsius(invalid)
            self.assertAlmostEqual(thermostat.read_fahrenheit(), 86)
