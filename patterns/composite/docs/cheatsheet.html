Composite — Компоновщик

Пример: слайды 32–41; задание: слайд 43 презентации «Лекция 3». Код ниже сохраняет операции со слайдов и дополняет их выводом дерева (describe()), удалением элементов и защитой от циклов. На слайдах эти вспомогательные детали опущены. Большое дерево на слайде 35 иллюстрирует вложенность; для задания используется небольшое дерево со слайда 43, приведённое ниже.

Проблема

В доме есть отдельные приборы, комнаты и этажи. Если хранить их как несвязанные типы, клиенту придётся разбирать структуру вручную: найти комнаты, перебрать приборы, отдельно обработать вложенные группы.

Нужна единая команда: выключить лампу, комнату или весь дом.

Решение и роли

Устройство выполняет команду само, а группа передаёт её детям. Среди детей могут быть как устройства, так и другие группы. Рекурсия проходит всё поддерево.

РольКласс
Component — общий интерфейсComponent: turn_on(), turn_off(), describe()
Leaf — конечный элементDevice
Composite — контейнер компонентовDeviceGroup
Дом (DeviceGroup)
└── Первый этаж (DeviceGroup)
    ├── Гостиная (DeviceGroup)
    │   ├── Лампа (Device)
    │   └── Телевизор (Device)
    └── Кухня (DeviceGroup)
        └── Чайник (Device)

house.turn_off() выключает все приборы в доме. living_room.turn_off() затрагивает только гостиную. lamp.turn_off() выключает только лампу. Клиент использует одинаковую операцию.

Только группа имеет add() и remove(): устройство не обязано поддерживать бессмысленную для него операцию добавления детей. Свойство children возвращает кортеж. Чтобы перенести компонент, сначала вызовите remove() у старой группы. Проверки в add() запрещают циклы, повторное добавление и двух родителей у одного компонента.

Применимость и ограничения

Composite подходит для иерархий «часть — целое»: каталогов, меню, учебных программ, групп элементов интерфейса. Клиенту не приходится отличать лист от контейнера при выполнении общей операции. При этом общий интерфейс должен иметь смысл для обоих видов компонентов; дерево также требует защиты от циклов.

Вопрос: почему список устройств без вложенных групп ещё не показывает возможности Composite?

Реализация: tree.py
"""Composite: дом → этаж → комната → устройство.

Component — Component; Device — Leaf; DeviceGroup — Composite.
Устройства и группы имеют общие команды; только группы хранят детей.
"""

from __future__ import annotations

from abc import ABC, abstractmethod


class Component(ABC):
    def __init__(self, name: str) -> None:
        self.name = name
        self._parent: DeviceGroup | None = None

    @abstractmethod
    def turn_on(self) -> None:
        """Включить компонент."""

    @abstractmethod
    def turn_off(self) -> None:
        """Выключить компонент."""

    @abstractmethod
    def describe(self, indent: int = 0) -> list[str]:
        """Вернуть строки для вывода дерева."""


class Device(Component):
    def __init__(self, name: str) -> None:
        super().__init__(name)
        self.is_on = False

    def turn_on(self) -> None:
        self.is_on = True

    def turn_off(self) -> None:
        self.is_on = False

    def describe(self, indent: int = 0) -> list[str]:
        state = "включено" if self.is_on else "выключено"
        return [f"{'  ' * indent}{self.name}: {state}"]


class DeviceGroup(Component):
    def __init__(self, name: str) -> None:
        super().__init__(name)
        self._children: list[Component] = []

    @property
    def children(self) -> tuple[Component, ...]:
        return tuple(self._children)

    def add(self, child: Component) -> None:
        if not isinstance(child, Component):
            raise TypeError("В группу можно добавить только Component")
        # Циклы привели бы к бесконечной рекурсии при выполнении команд.
        ancestor: Component | None = self
        while ancestor is not None:
            if ancestor is child:
                raise ValueError("Нельзя добавить группу в саму себя или своего потомка")
            ancestor = ancestor._parent
        if child._parent is not None:
            raise ValueError("Компонент уже в группе; сначала удалите его из неё")
        self._children.append(child)
        child._parent = self

    def remove(self, child: Component) -> None:
        self._children.remove(child)
        child._parent = None

    def turn_on(self) -> None:
        for child in self._children:
            child.turn_on()

    def turn_off(self) -> None:
        for child in self._children:
            child.turn_off()

    def describe(self, indent: int = 0) -> list[str]:
        lines = [f"{'  ' * indent}{self.name}/"]
        for child in self._children:
            lines.extend(child.describe(indent + 1))
        return lines

import unittest

from .. import tree
from ..tree import Device, DeviceGroup


class CompositeTests(unittest.TestCase):
    def test_nested_group_commands_reach_all_leaves(self):
        house = DeviceGroup("Дом")
        room = DeviceGroup("Комната")
        lamp = Device("Лампа")
        tv = Device("Телевизор")
        hall_light = Device("Прихожая")
        room.add(lamp)
        room.add(tv)
        house.add(room)
        house.add(hall_light)
        house.turn_on()
        self.assertTrue(all(d.is_on for d in (lamp, tv, hall_light)))
        room.turn_off()
        self.assertFalse(lamp.is_on)
        self.assertFalse(tv.is_on)
        self.assertTrue(hall_light.is_on)
        house.turn_off()
        self.assertFalse(hall_light.is_on)

    def test_leaf_and_empty_group_support_common_operations(self):
        for component in (Device("Лампа"), DeviceGroup("Пустая комната")):
            component.turn_on()
            self.assertTrue(component.describe())
            component.turn_off()
        self.assertFalse(component.children)

    def test_removed_device_can_move_and_no_longer_receives_old_commands(self):
        old = DeviceGroup("Старая комната")
        new = DeviceGroup("Новая комната")
        lamp = Device("Лампа")
        old.add(lamp)
        old.remove(lamp)
        old.turn_on()
        self.assertFalse(lamp.is_on)
        new.add(lamp)
        new.turn_on()
        self.assertTrue(lamp.is_on)

    def test_tree_rejects_self_and_ancestor_cycles(self):
        root = DeviceGroup("Дом")
        room = DeviceGroup("Комната")
        root.add(room)
        with self.assertRaises(ValueError):
            root.add(root)
        with self.assertRaises(ValueError):
            room.add(root)
        self.assertEqual(root.children, (room,))
        self.assertEqual(room.children, ())

    def test_component_cannot_have_two_parents_or_be_added_twice(self):
        first = DeviceGroup("Первая")
        second = DeviceGroup("Вторая")
        lamp = Device("Лампа")
        first.add(lamp)
        with self.assertRaises(ValueError):
            first.add(lamp)
        with self.assertRaises(ValueError):
            second.add(lamp)
        self.assertEqual(first.children, (lamp,))
        self.assertEqual(second.children, ())

    def test_invalid_add_and_remove_preserve_group(self):
        group = DeviceGroup("Комната")
        lamp = Device("Лампа")
        group.add(lamp)
        with self.assertRaises(TypeError):
            group.add("не устройство")
        with self.assertRaises(ValueError):
            group.remove(Device("Чужая лампа"))
        self.assertEqual(group.children, (lamp,))

import unittest
from unittest.mock import patch

from .. import tree
from ..tree import Device, DeviceGroup


class BuildHouseTaskTests(unittest.TestCase):
    def setUp(self):
        build_house = getattr(tree, "build_demo_house", None)
        self.assertTrue(
            callable(build_house),
            "ЗАДАНИЕ: добавьте build_demo_house() в tree.py (см. TASK.md)",
        )
        self.build_house = build_house
        self.house = build_house()

    def test_structure_matches_assignment(self):
        self.assertIsInstance(self.house, DeviceGroup)
        self.assertEqual(self.house.name, "Дом")
        self.assertEqual(len(self.house.children), 1)
        floor = self.house.children[0]
        self.assertIsInstance(floor, DeviceGroup)
        self.assertEqual(floor.name, "Первый этаж")
        self.assertEqual([c.name for c in floor.children], ["Гостиная", "Кухня"])
        living, kitchen = floor.children
        for room in (living, kitchen):
            self.assertIsInstance(room, DeviceGroup)
        self.assertEqual([d.name for d in living.children], ["Лампа", "Телевизор"])
        self.assertEqual([d.name for d in kitchen.children], ["Чайник"])
        for device in (*living.children, *kitchen.children):
            self.assertIsInstance(device, Device)
            self.assertFalse(device.is_on)

    def test_house_and_room_commands(self):
        living, kitchen = self.house.children[0].children
        self.house.turn_on()
        for device in (*living.children, *kitchen.children):
            self.assertTrue(device.is_on)
        living.turn_off()
        self.assertTrue(all(not d.is_on for d in living.children))
        self.assertTrue(all(d.is_on for d in kitchen.children))
        self.house.turn_off()
        self.assertTrue(all(not d.is_on for d in (*living.children, *kitchen.children)))

    def test_each_call_creates_independent_devices(self):
        self.house.turn_on()
        other = self.build_house()
        self.assertIsNot(other, self.house)

        def nodes(house):
            floor = house.children[0]
            living, kitchen = floor.children
            return (house, floor, living, kitchen, *living.children, *kitchen.children)

        self.assertTrue(set(map(id, nodes(self.house))).isdisjoint(map(id, nodes(other))))
        living, kitchen = other.children[0].children
        for device in (*living.children, *kitchen.children):
            self.assertFalse(device.is_on)

    def test_builder_uses_add_for_every_edge(self):
        edges = []
        original_add = DeviceGroup.add

        def record_add(group, child):
            edges.append((group, child))
            original_add(group, child)

        with patch.object(DeviceGroup, "add", record_add):
            house = self.build_house()

        def collect_edges(group):
            result = []
            for child in group.children:
                result.append((group, child))
                if isinstance(child, DeviceGroup):
                    result.extend(collect_edges(child))
            return result

        self.assertCountEqual(edges, collect_edges(house))
